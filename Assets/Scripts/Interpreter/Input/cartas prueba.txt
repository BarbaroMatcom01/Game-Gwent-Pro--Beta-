effect {
Name: "ReturnHand",
Action: (targets , context) =>
 {
    for( target in targets){
       var owner= target.Owner;
       var hand = context.HandOfPlayer(owner);
      hand.Push(target);
      context.Board.Remove(target);
  };
  }
}
