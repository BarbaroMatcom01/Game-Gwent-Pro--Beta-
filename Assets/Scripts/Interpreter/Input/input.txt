
effect {
Name: "ReturnHand",
Action: (targets , context) =>
 {
    for( target in targets){
       var owner= target.Owner;
    var    deck = context.DeckOfPlayer(owner);
      deck.Push(target);
      deck.Shuffle();
      context.Board.Remove(target);
  }
  }
}
card {
    Type: "Silver",
    Name: "Carta Interpretada",
    Faction: "Narnia",
    Power: 10,
    Range: "Ranged", 
    Description:" Carta creada en el compilador",
    
OnActivation: [
        { 
          Effect : {
              Name: "ReturnHand",
            },  
           Selector:
            {
               Single:false ,
               Source: "board", 
               Predicate: (unit) => unit.Name == "Fauno",
            }
      }
]
}
